cmake_minimum_required(VERSION 3.26)
project(nemotron_lifecycle_tests LANGUAGES CXX)
enable_testing()
add_executable(lifecycle_tests lifecycle.cpp)
target_compile_features(lifecycle_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(lifecycle_tests PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(lifecycle_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME sequential_and_repeated_utterances COMMAND lifecycle_tests 0)
add_test(NAME creation_and_decode_failure_cleanup COMMAND lifecycle_tests 1)
add_test(NAME finalization_and_drain_failure_cleanup COMMAND lifecycle_tests 2)
add_test(NAME cancel_then_reuse COMMAND lifecycle_tests 3)
add_test(NAME shutdown_and_exception_cleanup COMMAND lifecycle_tests 4)

# Optional model-free tests of the actual production adapter, using the pinned
# public header and a synthetic C ABI implementation; never link it into release.
set(NEMO_SPEECH_SOURCE_DIR "" CACHE PATH "Reviewed native source for boundary tests")
if(WIN32 AND EXISTS "${NEMO_SPEECH_SOURCE_DIR}/include/nemo_speech/asr.h")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(boundary_worker ../main.cpp fake_asr.cpp)
    target_include_directories(boundary_worker PRIVATE "${NEMO_SPEECH_SOURCE_DIR}/include")
    target_compile_features(boundary_worker PRIVATE cxx_std_20)
    target_compile_definitions(boundary_worker PRIVATE NEMO_SPEECH_ASR_BUILD NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
        target_compile_options(boundary_worker PRIVATE /W4 /WX /EHsc /utf-8)
    endif()
    add_test(NAME production_adapter_malformed_and_failure_matrix
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/boundary.py" $<TARGET_FILE:boundary_worker>)
    set_tests_properties(production_adapter_malformed_and_failure_matrix PROPERTIES TIMEOUT 120)
endif()
